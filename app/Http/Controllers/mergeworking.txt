<?php

namespace App\Http\Controllers;

use Barryvdh\DomPDF\Facade\Pdf;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use RuntimeException;
use setasign\Fpdi\Fpdi;
use setasign\Fpdi\PdfParser\PdfParserException;
use Throwable;

class DocumentPreviewController extends Controller
{
    public function preview(Request $request)
    {
        $key = array_key_first($request->all());
        $data = session()->get($key);

        if (!$data) {
            abort(404, 'Preview session expired or not found.');
        }

        $data = $this->normalizePreviewData($data);
        $temporaryFiles = [];

        $generatedPdf = $this->temporaryFile('generated_');
        $temporaryFiles[] = $generatedPdf;

        Pdf::loadView('pdf.document-preview', $data)
            ->setPaper([0, 0, 612.00, 936.00])
            ->save($generatedPdf);

        $filesToMerge = [$generatedPdf];
        foreach ($data['attachments'] ?? [] as $attachment) {
            $this->processAttachment($attachment, $filesToMerge, $temporaryFiles);
        }

        $pdfToShow = $generatedPdf;
        if (count($filesToMerge) > 1) {
            $mergedPdf = $this->temporaryFile('merged_');
            $temporaryFiles[] = $mergedPdf;

            $this->mergePdfs($filesToMerge, $mergedPdf);
            $pdfToShow = $mergedPdf;
        }

        session()->forget($key);

        register_shutdown_function(function () use ($temporaryFiles): void {
            foreach ($temporaryFiles as $file) {
                if (is_file($file)) {
                    @unlink($file);
                }
            }
        });

        $filename = $data['action'] === 'preview'
            ? 'document-preview.pdf'
            : ($data['documentNumber'] ?? 'Document') . '.pdf';

        return response()->file($pdfToShow, [
            'Content-Type' => 'application/pdf',
            'Content-Disposition' => 'inline; filename="' . $filename . '"',
        ]);
    }

    private function normalizePreviewData(array $data): array
    {
        $data['date_sent'] = $data['date_sent'] ?? now();

        foreach (['document', 'signatories', 'attachments', 'cfs'] as $field) {
            if (isset($data[$field]) && is_string($data[$field])) {
                $data[$field] = json_decode($data[$field], true);
            }
        }

        return $data;
    }

    private function processAttachment(array $attachment, array &$filesToMerge, array &$temporaryFiles): void
    {
        $fileUrl = $attachment['file_url'] ?? null;
        if (!$fileUrl) {
            return;
        }

        $path = public_path('storage/' . $fileUrl);
        if (!is_file($path)) {
            Log::warning('Skipping a missing document attachment.', ['attachment' => $fileUrl]);
            return;
        }

        $extension = strtolower(pathinfo($path, PATHINFO_EXTENSION));

        if ($extension === 'pdf') {
            try {
                (new Fpdi())->setSourceFile($path);
                $filesToMerge[] = $path;
            } catch (PdfParserException $exception) {
                // The original remains available through its attachment URL; it is only
                // omitted from this combined preview because free FPDI cannot parse it.
                Log::warning('Skipping an attachment that FPDI cannot merge.', [
                    'attachment' => $fileUrl,
                    'message' => $exception->getMessage(),
                ]);
            }

            return;
        }

        if (!in_array($extension, ['jpg', 'jpeg', 'png'], true)) {
            return;
        }

        $imagePdf = $this->temporaryFile('image_');
        $temporaryFiles[] = $imagePdf;

        Pdf::loadView('pdf.image-wrapper', ['imagePath' => $path])
            ->setPaper([0, 0, 612.00, 936.00])
            ->save($imagePdf);

        $filesToMerge[] = $imagePdf;
    }

    private function mergePdfs(array $files, string $outputPath): void
    {
        $pdf = new Fpdi();

        foreach ($files as $file) {
            try {
                $pageCount = $pdf->setSourceFile($file);

                for ($pageNo = 1; $pageNo <= $pageCount; $pageNo++) {
                    $templateId = $pdf->importPage($pageNo);
                    $size = $pdf->getTemplateSize($templateId);

                    $pdf->AddPage($size['orientation'], [$size['width'], $size['height']]);
                    $pdf->useTemplate($templateId);
                }
            } catch (Throwable $exception) {
                Log::warning('Skipping an attachment during PDF merge.', [
                    'file' => $file,
                    'message' => $exception->getMessage(),
                ]);
            }
        }

        $pdf->Output('F', $outputPath);
    }

    private function temporaryFile(string $prefix): string
    {
        $path = tempnam(sys_get_temp_dir(), $prefix);

        if ($path === false) {
            throw new RuntimeException('Unable to create a temporary PDF file.');
        }

        return $path;
    }
}
